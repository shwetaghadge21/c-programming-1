#include <stdio.h>

void main()
{
    int sensor1, sensor2, sensor3, sensor4, mul;

    printf("Enter value of sensor1: ");
    scanf("%d", &sensor1);

    printf("Enter value of sensor2: ");
    scanf("%d", &sensor2);

    printf("Enter value of sensor3: ");
    scanf("%d", &sensor3);

    printf("Enter value of sensor4: ");
    scanf("%d", &sensor4);

    mul = sensor1 * sensor2 * sensor3 * sensor4;

    printf("Multiplication = %d", mul);
}
